@bot.group(invoke_without_command=True)
async def vm(ctx):
    data = load_data(ctx.guild.id)
    msgs = data["voicemail"]["messages"]
    user_msgs = [m for m in msgs if m["to_id"] == str(ctx.author.id) and not m.get("deleted")]
    if not user_msgs:
        return await ctx.send("📞 No voicemails. Inbox empty.")
    embed = discord.Embed(title=f"📞 {ctx.author.display_name}'s Voicemail Inbox", color=0xFF4500)
    for m in user_msgs[-10:][::-1]:
        embed.add_field(
            name=f"📞 #{m['id']} • From: {m['from']} • {m['time']}",
            value=f"💬 _{m['text'][:100]}_",
            inline=False
        )
    await ctx.send(embed=embed)

@vm.command()
async def leave(ctx, user: discord.Member, *, text: str):
    data = load_data(ctx.guild.id)
    mid = data["voicemail"]["next_id"]
    data["voicemail"]["next_id"] += 1
    data["voicemail"]["messages"].append({
        "id": mid, "from": str(ctx.author), "from_id": str(ctx.author.id),
        "to": str(user), "to_id": str(user.id), "text": text,
        "time": datetime.now().strftime("%Y-%m-%d %H:%M"),
        "listened": False, "deleted": False
    })
    save_data(ctx.guild.id, data)
    embed = discord.Embed(
        title="📞 Voicemail Left",
        description=f"💬 *\"{text}\"*",
        color=0xFF4500
    )
    embed.add_field(name="To", value=user.mention, inline=True)
    embed.add_field(name="From", value=ctx.author.mention, inline=True)
    embed.set_footer(text=f"Voicemail #{mid}")
    await ctx.send(embed=embed)

@vm.command()
async def listen(ctx, mid: int):
    data = load_data(ctx.guild.id)
    msg = next((m for m in data["voicemail"]["messages"] if m["id"] == mid), None)
    if not msg or msg["to_id"] != str(ctx.author.id):
        return await ctx.send("❌ Voicemail not found.")
    msg["listened"] = True
    save_data(ctx.guild.id, data)
    embed = discord.Embed(
        title=f"📞 Voicemail #{mid}",
        description=f"💬 *\"{msg['text']}\"*",
        color=0xFF4500
    )
    embed.add_field(name="From", value=msg['from'], inline=True)
    embed.add_field(name="When", value=msg['time'], inline=True)
    await ctx.send(embed=embed)

@bot.group(invoke_without_command=True)
async def radio(ctx):
    data = load_data(ctx.guild.id)
    if not data["amfm"]["stations"]:
        return await ctx.send("📻 No stations. `!radio add <freq> <band> <name>`")
    embed = discord.Embed(title="📻 AM/FM Stations", color=0xFF6B35)
    for freq, s in data["amfm"]["stations"].items():
        playing = "🔴 LIVE" if data["amfm"]["current"].get(freq) else "⚫ Off Air"
        embed.add_field(
            name=f"📻 {s['band']} {freq} — {s['name']}",
            value=f"{playing}\n{', '.join(s.get('shows', [])) or 'No shows'}",
            inline=False
        )
    await ctx.send(embed=embed)

@radio.command()
async def add(ctx, freq: str, band: str, *, name: str):
    data = load_data(ctx.guild.id)
    band = band.upper()
    if band not in AMFM_BANDS:
        return await ctx.send("❌ Band must be AM or FM.")
    if freq in data["amfm"]["stations"]:
        return await ctx.send("⚠️ Frequency taken.")
    data["amfm"]["stations"][freq] = {
        "freq": freq, "band": band, "name": name,
        "shows": [], "added_by": str(ctx.author)
    }
    save_data(ctx.guild.id, data)
    await ctx.send(f"📻 Added **{band} {freq}** — {name}")

@radio.command()
async def tune(ctx, freq: str):
    data = load_data(ctx.guild.id)
    s = data["amfm"]["stations"].get(freq)
    if not s:

@feed.command()
async def post(ctx, *, text: str):
    data = load_data(ctx.guild.id)
    pid = data["feed"]["next_id"]
    data["feed"]["next_id"] += 1
    data["feed"]["posts"].append({"id": pid, "author": str(ctx.author), "text": text, "time": datetime.now().strftime("%H:%M")})
    data["feed"]["reactions"][str(pid)] = {}
    data["feed"]["comments"][str(pid)] = []
    save_data(ctx.guild.id, data)
    await ctx.send(f"📰 Posted #{pid}: {text}")

@feed.command()
async def view(ctx, pid: int):
    data = load_data(ctx.guild.id)
    p = next((p for p in data["feed"]["posts"] if p["id"] == pid), None)
    if not p:
        return await ctx.send("❌ Not found.")
    await ctx.send(f"📰 **#{p['id']}** by {p['author']}\n{p['text']}")
